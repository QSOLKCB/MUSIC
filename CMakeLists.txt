cmake_minimum_required(VERSION 3.20)
project(MUSIC VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(MUSIC_BUILD_GUI "Build the native GTK desktop" ON)
option(MUSIC_SANITIZE "Run core tests with address/undefined behaviour sanitizers" OFF)
add_library(music-core STATIC src/music.c)
target_include_directories(music-core PUBLIC src)
if(MSVC)
  target_compile_definitions(music-core PUBLIC _CRT_SECURE_NO_WARNINGS)
  target_compile_options(music-core PRIVATE /W4)
else()
  target_compile_options(music-core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(MUSIC_SANITIZE AND NOT MSVC)
  target_compile_options(music-core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(music-core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(music-cli src/cli.c)
target_link_libraries(music-cli PRIVATE music-core)
enable_testing()
add_executable(music-core-tests tests/test_core.c)
target_link_libraries(music-core-tests PRIVATE music-core)
add_test(NAME core COMMAND music-core-tests)
add_executable(music-audio-capacity-tests tests/test_audio_capacity.c)
target_include_directories(music-audio-capacity-tests PRIVATE tests/audio-stub)
target_link_libraries(music-audio-capacity-tests PRIVATE music-core)
add_test(NAME audio-capacity COMMAND music-audio-capacity-tests)
if(MUSIC_BUILD_GUI)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4>=4.10)
  set(MINIAUDIO_HEADER "${CMAKE_BINARY_DIR}/dependencies/miniaudio.h")
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/dependencies")
  if(NOT EXISTS "${MINIAUDIO_HEADER}")
    file(DOWNLOAD
      "https://raw.githubusercontent.com/mackron/miniaudio/877ea4a7c2601c7b0f2a44e83dfac42e6470fea6/miniaudio.h"
      "${MINIAUDIO_HEADER}"
      EXPECTED_HASH SHA256=6b376c7fbb1352ba1e0e432ab66069b335ea9405e45cb6c30b97ef0f3c9c3e0e
      TLS_VERIFY ON)
  endif()
  file(SHA256 "${MINIAUDIO_HEADER}" AUDIO_HASH)
  if(NOT AUDIO_HASH STREQUAL "6b376c7fbb1352ba1e0e432ab66069b335ea9405e45cb6c30b97ef0f3c9c3e0e")
    message(FATAL_ERROR "Pinned miniaudio header checksum mismatch")
  endif()
  add_executable(music-app MACOSX_BUNDLE src/main.c src/audio.c)
  set_target_properties(music-app PROPERTIES OUTPUT_NAME music MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/Info.plist.in")
  target_include_directories(music-app PRIVATE "${CMAKE_BINARY_DIR}/dependencies")
  target_link_libraries(music-app PRIVATE music-core PkgConfig::GTK4 ${CMAKE_DL_LIBS})
  if(UNIX)
    find_package(Threads REQUIRED)
    target_compile_definitions(music-app PRIVATE _POSIX_C_SOURCE=200809L)
    target_link_libraries(music-app PRIVATE Threads::Threads m)
  endif()
  if(APPLE)
    target_link_libraries(music-app PRIVATE "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox")
  endif()
  if(WIN32)
    set_target_properties(music-app PROPERTIES WIN32_EXECUTABLE TRUE)
    add_executable(music-runtime-check tests/test_windows_runtime.c)
    target_link_libraries(music-runtime-check PRIVATE PkgConfig::GTK4)
  endif()
  install(TARGETS music-app RUNTIME DESTINATION . BUNDLE DESTINATION .)
endif()
install(TARGETS music-cli RUNTIME DESTINATION .)
install(FILES LICENSE README.md docs/MINIAUDIO-LICENSE.txt DESTINATION .)

add_executable(music-render-tests tests/test_render.c tests/render_reference.c)
target_link_libraries(music-render-tests PRIVATE music-core)
add_test(NAME render-equivalence COMMAND music-render-tests)

# Packaging helpers are build-time tooling, never runtime dependencies.
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME windows-packaging-helpers
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_windows_packaging.py")
endif()
